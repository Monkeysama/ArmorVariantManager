-- 枪械候选条件探针(临时工具,测完就删)。独立脚本,不 require 差分管理器的任何模块。
--
-- 为什么不用原来那个「条件判断」面板:这一批候选**全是瞬时标志**(空枪击发、泵动上膛、
-- 快速装填、枪口顶墙),只显示"当前为真"的话,亮一帧就过去了,眼睛跟不上。
-- 所以这里做**闩锁**:记住每条最后一次为真是多少帧前、以及一共亮过多少帧、
-- 单次最长连续亮多少帧 —— 做完动作再回来看面板就行。
--
-- 只用局部变量(REFramework 的 autorun 共享同一个 Lua 状态)。

local version = "0.1.0"

-- 候选(要测的) + 参照(已实测过语义的,用来对齐时间轴)
local CANDIDATES = {
    { m = "get_IsShootDryFire",        note = "空枪击发?" },
    { m = "get_IsPumpAction",          note = "泵动上膛?" },
    { m = "get_IsExReload",            note = "快速装填?" },
    { m = "get_IsGunWallHit",          note = "枪口顶墙?" },
    { m = "get_IsWaitingGenerateShell", note = "等待生成弹壳?" },
    { m = "get_IsShootEnable",         note = "可射击?" },
    { m = "get_RequestPumpAction",     note = "请求泵动?" },
}

local REFERENCE = {
    { m = "get_IsPullTrigger",  note = "扣扳机(空枪也算)" },
    { m = "get_IsShooting",     note = "真打出子弹" },
    { m = "get_IsReloading",    note = "装填中" },
    { m = "get_IsAiming",       note = "瞄准中移动" },
    { m = "get_IsWheel",        note = "站定举枪/举刀" },
}

local NUMBERS = {
    { m = "get_ThreeBurstCounter", note = "三连发计数" },
}

-- ---------- 引擎访问 ----------

local function player_context()
    local cm = sdk.get_managed_singleton("chainsaw.CharacterManager")
    if not cm then return nil end
    local ok, ctx = pcall(function() return cm:call("getPlayerContextRef") end)
    if ok and ctx and sdk.is_managed_object(ctx) then return ctx end
    return nil
end

-- 方法存在性先反射确认,查不到就记进 missing、之后再也不调 ——
-- 直接 call 一个不存在的方法会每帧往日志里灌错误。
local missing = {}
local present = {}
local function has_method(ctx, name)
    if missing[name] then return false end
    if present[name] then return true end
    local td = nil
    local ok, t = pcall(function() return ctx:get_type_definition() end)
    if ok then td = t end
    while td do
        local ok_m, found = pcall(function() return td:get_method(name) end)
        if ok_m and found then present[name] = true; return true end
        local ok_p, parent = pcall(function() return td:get_parent_type() end)
        if not ok_p then break end
        td = parent
    end
    missing[name] = true
    return false
end

local function read(ctx, name)
    if not has_method(ctx, name) then return nil end
    local ok, v = pcall(function() return ctx:call(name) end)
    if ok then return v end
    return nil
end

-- ---------- 闩锁 ----------

local frame = 0
local st = {}   -- [method] = { true_frames, last_true_frame, streak, max_streak, seen, cur }

local function state_of(m)
    local s = st[m]
    if not s then
        s = { true_frames = 0, last_true_frame = nil, streak = 0, max_streak = 0, seen = false, cur = nil }
        st[m] = s
    end
    return s
end

local function sample()
    local ctx = player_context()
    if not ctx then return end
    frame = frame + 1
    local lists = { CANDIDATES, REFERENCE }
    for _, list in ipairs(lists) do
        for _, e in ipairs(list) do
            local s = state_of(e.m)
            local v = read(ctx, e.m)
            s.cur = v
            if v == true then
                s.seen = true
                s.true_frames = s.true_frames + 1
                s.last_true_frame = frame
                s.streak = s.streak + 1
                if s.streak > s.max_streak then s.max_streak = s.streak end
            else
                s.streak = 0
            end
        end
    end
    for _, e in ipairs(NUMBERS) do
        local s = state_of(e.m)
        local v = read(ctx, e.m)
        s.cur = v
        if type(v) == "number" then
            if s.min_v == nil or v < s.min_v then s.min_v = v end
            if s.max_v == nil or v > s.max_v then s.max_v = v end
        end
    end
end

re.on_frame(function()
    pcall(sample)
end)

-- ---------- 界面 ----------

local last_error = nil
local tree_open = false

local function tree_pop()
    if tree_open then
        tree_open = false
        pcall(imgui.tree_pop)
    end
end

-- 颜色只用各通道对称的值,免得赌 ARGB / ABGR
local GREEN = 0xFF00FF00
local GREY  = 0xFF808080

local function row(e)
    local s = state_of(e.m)
    local ago = s.last_true_frame and (frame - s.last_true_frame) or nil
    local label = string.format("%-28s %s", e.m:gsub("^get_", ""), e.note or "")
    if s.cur == true then
        imgui.text_colored(label .. "   ● NOW", GREEN)
    elseif s.cur == nil then
        imgui.text_colored(label .. "   (读不到 / not readable)", GREY)
    elseif s.seen then
        imgui.text(string.format("%s   亮过 %d 帧 / 最长连亮 %d / 上次 %d 帧前",
            label, s.true_frames, s.max_streak, ago or -1))
    else
        imgui.text_colored(label .. "   —— 没见亮过", GREY)
    end
end

local function draw()
    local ok_tree, open = pcall(imgui.tree_node, "枪械条件探针 (RE4) v" .. version)
    if not ok_tree or not open then return end
    tree_open = true

    if last_error then imgui.text("绘制出错(已保护): " .. last_error) end

    local ctx = player_context()
    if not ctx then
        imgui.text("未找到当前控制角色")
        tree_pop()
        return
    end

    imgui.text("采样帧数: " .. frame)
    if imgui.button("清空统计###gunprobe_reset") then
        st = {}
        frame = 0
    end

    imgui.separator()
    imgui.text("候选:")
    for _, e in ipairs(CANDIDATES) do row(e) end

    imgui.separator()
    imgui.text("参照(语义已知):")
    for _, e in ipairs(REFERENCE) do row(e) end

    imgui.separator()
    imgui.text("数值:")
    for _, e in ipairs(NUMBERS) do
        local s = state_of(e.m)
        imgui.text(string.format("%-28s %s   当前=%s  区间=[%s, %s]",
            e.m:gsub("^get_", ""), e.note or "",
            tostring(s.cur), tostring(s.min_v), tostring(s.max_v)))
    end

    imgui.separator()
    imgui.text("怎么测:1) 清空统计  2) 做一个动作(例如把手枪打空后再按射击)  3) 回来看哪条亮过、亮了几帧")
    tree_pop()
end

re.on_draw_ui(function()
    local ok, err = pcall(draw)
    if ok then
        last_error = nil
    else
        local msg = tostring(err)
        if last_error ~= msg then
            pcall(function() log.error("[GunProbe] " .. msg) end)
        end
        last_error = msg
    end
    tree_pop()
end)
