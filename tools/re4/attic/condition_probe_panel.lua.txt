-- ============================================================
-- 条件判断(只读诊断面板)
--
-- 目的:在游戏里摆出想要的姿势/场合,看哪一条候选条件变绿 —— 先确认语义,再决定实装。
-- 这比"照名字猜 → 实装 → 发现不对 → 回退"快得多(README 里那几个反例都是这么踩的)。
--
-- 颜色只用 0xFF00FF00 / 灰阶这类各通道对称或相等的值:REFramework 的
-- imgui.text_colored 到底是 ARGB 还是 ABGR 不必赌,这些值两种解读下都是同一个颜色。
-- 读不到的条目额外带一句文字说明,不靠颜色区分"假"和"读不到"。
-- ============================================================

local probe_hp_buf = "1"

local PROBE_COLOR_HIT  = 0xFF00FF00
local PROBE_COLOR_MISS = 0xFF9A9A9A
local PROBE_COLOR_NA   = 0xFF5A5A5A

local function colored_text(text, color)
    if not pcall(imgui.text_colored, text, color) then imgui.text(text) end
end

local function render_condition_probe(context)
    local lang = global_config.language or "en"

    -- 已实装成真条件的打个星号:这份清单同时回答"能测什么"和"现在能用什么"
    local registered = {}
    for _, def in ipairs(Registry.all()) do
        if def.method then registered[def.method] = true end
    end

    -- 先全部取一遍值再画:组标题里要带"这组命中几条",而标题必须在内容之前画出来。
    -- 顺便保证每个条目每帧只读一次。
    local results, hit_total, read_total, all_total = {}, 0, 0, 0
    for gi, group in ipairs(Probe.BOOL_GROUPS) do
        local rows, hits = {}, 0
        for _, item in ipairs(group.items) do
            local state = Probe.probe_bool(context, item)
            table.insert(rows, { item = item, state = state })
            all_total = all_total + 1
            if state ~= "na" then read_total = read_total + 1 end
            if state == "on" then hits = hits + 1 end
        end
        hit_total = hit_total + hits
        results[gi] = { group = group, rows = rows, hits = hits }
    end

    imgui.text(string.format("%s %d / %d      %s %d",
        T("probe_hit"), hit_total, read_total, T("probe_unreadable"), all_total - read_total))
    imgui.text(T("probe_hint"))
    imgui.separator()

    for _, r in ipairs(results) do
        local header = string.format("%s  [%d/%d]###probe_%s", T(r.group.key), r.hits, #r.rows, r.group.key)
        if ui_tree(header) then
            for _, row in ipairs(r.rows) do
                local label = Probe.label(row.item, lang)
                if registered[row.item.method] then label = label .. " *" end
                if row.state == "on" then
                    colored_text("  " .. label, PROBE_COLOR_HIT)
                elseif row.state == "off" then
                    colored_text("  " .. label, PROBE_COLOR_MISS)
                else
                    colored_text("  " .. label .. "  (" .. T("probe_na") .. ")", PROBE_COLOR_NA)
                end
            end
            ui_tree_pop()
        end
    end

    -- 取值类不做绿色判定:没有阈值就谈不上"符合",所以直接把当前值摊开给用户看
    if ui_tree(T("probe_g_values") .. "###probe_values") then
        for _, item in ipairs(Probe.VALUE_ITEMS) do
            local label = Probe.label(item, lang)
            if registered[item.method] then label = label .. " *" end
            local v = Probe.probe_value(context, item)
            if v then
                imgui.text("  " .. label .. " = " .. v)
            else
                colored_text("  " .. label .. "  (" .. T("probe_na") .. ")", PROBE_COLOR_NA)
            end
        end
        ui_tree_pop()
    end

    -- 改血测试。放在取值组旁边是因为它只服务于一件事:把 HP 推到某个值,
    -- 看「血量分级」在哪一档翻(实测 Fine(0) 会直接跳到 Danger(2),中间的 1 没见过)。
    -- 刻意做成"填数字 + 点写入"而不是滑条:滑条会在拖动过程中连续写 HP,
    -- 一路扫过濒死/死亡阈值,把要观察的现象自己搅掉。
    imgui.separator()
    imgui.text(T("probe_hp_test"))
    local ch_hp, hv = imgui.input_text("###probe_hp_input", probe_hp_buf)
    if ch_hp then probe_hp_buf = hv end
    imgui.same_line()
    if imgui.button(T("probe_hp_set") .. "###probe_hp_apply") then
        local n = tonumber(probe_hp_buf)
        if n then Condition_HP.set_absolute(context, n) end
    end

    imgui.text(T("probe_impl_hint"))
end
